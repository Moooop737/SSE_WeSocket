01 · 结论与边界

这一格只换“回答怎么长”，不换“回答怎么来”

链路、检索、会话都不动：只把 assistant 的那段 text 从纯文本改成一棵 Markdown 树，并把引用、光标、图这三样接回去。

01 / OWNS

本格拥有的

markdown-answer.tsx（渲染入口、组件表、标号插件、光标）、citations.ts（Markdown 感知的引用解析）、mermaid-guard.ts 与 mermaid-diagram.tsx（图 + 导出 JPEG）。

02 / NOT OWNS

不归本格的

表格的「下载 CSV」、```csv 围栏成表、整条回答导出 .md —— 都是 4.5（页内带 4.5 徽章）。语法高亮、KaTeX、脚注仍在 Deferred；回答改成 JSON part 是 P4。

03 / SAFETY

安全默认

不解析 raw HTML；链接 target=_blank rel=noopener noreferrer；图片只渲染 alt、不发起请求；Mermaid 用 strict + htmlLabels:false，加源头与产物两道理。

02 · 全局地图

一段文字要经过四站才变成 DOM

每帧都是同一条路：解析 → 改写 → 转树 → 组件渲染。引用编号与渲染共用同一个解析器，所以不会出现“正文里是第 2 条、右栏里是第 3 条”。

  1. remark 解析：remark-parse + remark-gfm（表格、任务列表）+ remark-breaks（单换行即硬换行）。
  2. 标号改写：插件把正文 text 节点里的 [n] / [8 位 id] 换成 [n](citation:n) 链接；代码节点不遍历。
  3. mdast → hast → JSX：react-markdown 用模块级 COMPONENTS 渲染 a / pre / table / img。
  4. 引用解析：citedHits() 再跑一次同一个解析器取标号顺序，右栏与气泡共用这一份编号。

30/** 正文里按出现顺序取出的标号（带方括号），代码与图片 alt 不算 */
31function citationTokens(Text: string): string[] {
32 const Out: string[] = []
33 collect(parser.parse(Text) as unknown as MdNode, Out)
34 return Out
35}
36
37function collect(Node: MdNode, Out: string[]): void {
38 if (Node.type === 'code' || Node.type === 'inlineCode' || Node.type === 'html') return
39 if (Node.type === 'link') {
40 /* 纯数字文字的链接 = 旧口径的引用标号；其余链接整段跳过（里面的方括号不是引用） */
41 const Only = Node.children?.length === 1 ? Node.children[0] : null
42 if (Only?.type === 'text' && Only.value !== undefined && NUMERIC_LINK.test(Only.value)) Out.push(`[${Only.value}]`)
43 return
44 }
45 if (Node.type === 'text') {
46 if (Node.value === undefined) return
47 for (const Match of Node.value.matchAll(MARKER)) Out.push(Match[0])
48 return
49 }
50 for (const Child of Node.children ?? []) collect(Child, Out)
51}

53/* 两种标号都认：`[2]`（名次）与 `[96ff38b8]`（模型自己截的 chunkId 前 8 位）。
54 8 位 id 先在本条命中里找，再往更早几轮的命中里找；数字名次只在本条命中里找。
55 对不上的不认 —— 不假装点亮一条它没引的命中。 */
56function hitFor(Token: string, Hits: RetrievedHit[], Known: RetrievedHit[], AllHits: RetrievedHit[]): RetrievedHit | null {
57 /* 名次只有 1–2 位；6–8 位即便全是数字也是 chunkId 前缀，不能当名次去查下标 */
58 if (Token.length <= 2) {
59 const Index = Number(Token) - 1
60 return Hits[Index] ?? null
61 }
62 const byId = (List: RetrievedHit[]) => List.find((Hit) => Hit.chunkId.toLowerCase().startsWith(Token.toLowerCase()))
63 const Own = byId(AllHits)
64 if (Own) return Own
65 /* 本轮检索过就只认本轮：一个对不上本轮命中的 id 更可能是模型记混了旧块，
66 往历史里兜会把它说成"本轮引用"。跨轮只服务「这一轮压根没调工具」的那种回答。 */
67 return AllHits.length === 0 ? (byId(Known) ?? null) : null
68}
69
70/** 一条回答**引用到的**命中，按首次出现顺序去重 —— 这个顺序就是引用编号 1..N 的依据（2026-10-20 Lynx 定：
71 * 不跟召回名次走，否则正文会出现 2、3、6、8 这种跳号）。右栏「引用」列的就是这一批，不再列工具的整份输出。 */
72function citedHits(Text: string, Hits: RetrievedHit[], Known: RetrievedHit[] = [], AllHits: RetrievedHit[] = Hits): RetrievedHit[] {
73 const Seen = new Map<string, RetrievedHit>()
74 for (const Marker of citationTokens(Text)) {
75 const Hit = hitFor(Marker.slice(1, -1), Hits, Known, AllHits)
76 if (Hit && !Seen.has(Hit.chunkId)) Seen.set(Hit.chunkId, Hit)
77 }
78 return [...Seen.values()]
79}

26const UNSAFE_SVG: { Pattern: RegExp; Why: string }[] = [
27 { Pattern: /<\s*(foreignObject|img|image)\b/i, Why: '产物含可执行或可加载标签' },
28 { Pattern: /\son[a-z]+\s*=/i, Why: '产物含事件处理属性' },
29 { Pattern: /javascript:/i, Why: '产物含 javascript: 协议' },
30 /* 合法的 marker 引用是 url(#id)；只有远程 / data: 才值得拦 */
31 { Pattern: /url\(\s*['"]?(?:https?:|\/\/|data:)/i, Why: '产物含外部 url()' },
32]

35 for (const { Pattern, Why } of UNSAFE_SVG) if (Pattern.test(Svg)) return Why
36 return null
